cmake_minimum_required(VERSION 3.20)
project(pvd_symbol_types LANGUAGES CXX)
set(DIA_SDK "" CACHE PATH "DIA SDK directory containing include/dia2.h")
if(NOT DIA_SDK AND DEFINED ENV{VSINSTALLDIR})
    set(DIA_SDK "$ENV{VSINSTALLDIR}/DIA SDK")
endif()
if(NOT EXISTS "${DIA_SDK}/include/dia2.h")
    message(FATAL_ERROR "Set -DDIA_SDK to an installed Visual Studio DIA SDK directory")
endif()
add_executable(pvd-symbol-types symbol_types.cpp)
target_compile_features(pvd-symbol-types PRIVATE cxx_std_17)
target_include_directories(pvd-symbol-types PRIVATE "${DIA_SDK}/include")
target_link_libraries(pvd-symbol-types PRIVATE ole32 oleaut32)
target_compile_definitions(pvd-symbol-types PRIVATE UNICODE _UNICODE NOMINMAX)
